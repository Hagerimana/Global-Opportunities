# Backend dependencies for the Global Opportunities API.
#
# Versions are pinned to compatible ranges so a rebuild six months from now
# still installs a stack that works together, while still allowing patch
# releases.

# Web framework and the REST layer built on top of it.
Django>=5.1,<5.3
djangorestframework>=3.15,<3.17

# JWT authentication: token issue, refresh, and DRF authentication class.
djangorestframework-simplejwt>=5.3,<5.6

# Query-parameter filtering and search for the opportunity discovery endpoints.
django-filter>=24.3,<26

# Lets the Next.js frontend (port 3000) call this API (port 8000) from a browser.
django-cors-headers>=4.6,<5

# PostgreSQL driver. The [binary] extra ships prebuilt wheels so no compiler is
# needed on the target machine.
psycopg[binary]>=3.2,<4

# Asynchronous task queue for notifications, deadline reminders, and AI work.
# The [redis] extra installs the Redis client Celery needs.
celery[redis]>=5.4,<6

# Object storage backend for uploaded documents (S3 and compatible services).
# NOTE: installed but not yet wired up -- settings.py has no STORAGES section,
# so uploads currently go to the local MEDIA_ROOT.
django-storages[s3]>=1.14,<2

# Reads backend/.env. Imported by config/settings.py, so it is required at
# runtime -- but it is still missing from this file. Add it before deploying to
# a clean environment, or settings import will fail with ModuleNotFoundError.
# Environment configuration.
python-dotenv>=1.0,<2

# Country reference data.
pycountry

# Read and validate PDF structure in the project-upload workflow.
pypdf

# Image manipulation and reader watermarks.
Pillow

# Render uploaded PDF pages into private images.
pypdfium2

python-decouple
